Usage

import { createComicImpact } from './effects/comic-impact';
import './effects/comic-impact/effect.css';

const word = document.querySelector<HTMLElement>('#impact-word')!;

const impact = createComicImpact(word, {
  shape: 'starburst',
  fill: '#ffd23f',
  ink: '#12100e',
  inkWidth: 4,
  points: 12,
  irregularity: 0.35,
  speedLines: 14,
  offset: 6,
  pop: 0.45,
  shake: 3,
  rotation: -8,
  trigger: 'inview',
  duration: 900,
  seed: 1,
});

impact.replay();
impact.setOptions({ shape: 'spike', fill: '#ef476f' });
impact.destroy();
