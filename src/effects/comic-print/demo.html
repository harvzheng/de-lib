Usage

import { createComicPrint } from './effects/comic-print';
import './effects/comic-print/effect.css';

const panel = document.querySelector<HTMLElement>('#panel')!;

const print = createComicPrint(panel, {
  palette: 'newsprint',
  dotSize: 4,
  halftone: 0.55,
  screenAngle: 15,
  levels: 5,
  misregistration: 1.5,
  grain: 0.4,
  paper: '#f4ecd8',
  roughness: 0.3,
  contrast: 1.15,
});

print.setOptions({ palette: 'sunday' });
print.destroy();
