Drop it in

import './effect.css';
import { createFilmGrainVideo } from './index';

const film = createFilmGrainVideo(background, {
  src: '/media/footage.mp4',
  poster: '/media/poster.jpg',
  fps: 16,
  look: 'kodak-gold-200',
});

film.setOptions({ grain: 1.1, speed: 0.8 });
await film.setSource(fileUrl);
film.destroy();
